cmake_minimum_required(VERSION 3.25)

project(KosmosEngine
    VERSION 0.1.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)

find_package(Vulkan REQUIRED)

add_subdirectory(ThirdParty)
add_subdirectory(Engine)
add_subdirectory(Sandbox)

find_program(DXC_EXECUTABLE
    NAMES dxc.exe
    HINTS "$ENV{VULKAN_SDK}/Bin"
    REQUIRED
)

set(SHADER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Shaders")
set(SHADER_OUTPUT_DIR "${CMAKE_BINARY_DIR}/Shaders")

set(FORWARD_LIT_VERT_SHADER "${SHADER_SOURCE_DIR}/ForwardLit.vert.hlsl")

set(PBR_LIGHTING_INCLUDE "${SHADER_SOURCE_DIR}/PBRLighting.hlsli")
set(DEFERRED_LIGHTING_FRAG_SHADER "${SHADER_SOURCE_DIR}/DeferredLighting.frag.hlsl")
set(SSAO_FRAG_SHADER "${SHADER_SOURCE_DIR}/SSAO.frag.hlsl")

set(FORWARD_LIT_VERT_SPV "${SHADER_OUTPUT_DIR}/ForwardLit.vert.spv")
set(FORWARD_LIT_FRAG_SPV "${SHADER_OUTPUT_DIR}/ForwardLit.frag.spv")
set(FULLSCREEN_VERT_SPV "${SHADER_OUTPUT_DIR}/Fullscreen.vert.spv")
set(FULLSCREEN_FRAG_SPV "${SHADER_OUTPUT_DIR}/Fullscreen.frag.spv")
set(SHADOW_DEPTH_VERT_SPV "${SHADER_OUTPUT_DIR}/ShadowDepth.vert.spv")
set(SKYBOX_VERT_SPV "${SHADER_OUTPUT_DIR}/Skybox.vert.spv")
set(SKYBOX_FRAG_SPV "${SHADER_OUTPUT_DIR}/Skybox.frag.spv")
set(ENVIRONMENT_PREFILTER_COMP_SPV "${SHADER_OUTPUT_DIR}/EnvironmentPrefilter.comp.spv")
set(BLOOM_DOWNSAMPLE_COMP_SPV "${SHADER_OUTPUT_DIR}/BloomDownsample.comp.spv")
set(BLOOM_UPSAMPLE_COMP_SPV "${SHADER_OUTPUT_DIR}/BloomUpsample.comp.spv")

set(G_BUFFER_FRAG_SPV "${SHADER_OUTPUT_DIR}/GBuffer.frag.spv")
set(DEFERRED_LIGHTING_FRAG_SPV "${SHADER_OUTPUT_DIR}/DeferredLighting.frag.spv")
set(SSAO_FRAG_SPV "${SHADER_OUTPUT_DIR}/SSAO.frag.spv")

add_custom_command(
    OUTPUT ${FORWARD_LIT_VERT_SPV}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUTPUT_DIR}
    COMMAND ${DXC_EXECUTABLE} -spirv -T vs_6_0 -E main -Fo ${FORWARD_LIT_VERT_SPV} ${FORWARD_LIT_VERT_SHADER}
    DEPENDS ${FORWARD_LIT_VERT_SHADER}
    VERBATIM
)

add_custom_command(
    OUTPUT ${FORWARD_LIT_FRAG_SPV}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_OUTPUT_DIR}
    COMMAND ${DXC_EXECUTABLE} -spirv -T ps_6_0 -E main -I ${SHADER_SOURCE_DIR} -Fo ${FORWARD_LIT_FRAG_SPV} ${FORWARD_LIT_FRAG_SHADER}
    DEPENDS ${FORWARD_LIT_FRAG_SHADER} ${PBR_LIGHTING_INCLUDE}
    VERBATIM
)

target_compile_definitions(KosmosEngine
    PRIVATE
